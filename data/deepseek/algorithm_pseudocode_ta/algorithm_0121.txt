def TrainPCA(S):
    # Step 1: Combine spectra into matrix
    X = np.array(S)
    # Step 2: Mean-center the data
    X_centered = X - np.mean(X, axis=0)
    # Step 3: Compute covariance matrix
    C = (X_centered.T @ X_centered) / (len(S) - 1)
    # Step 4: Diagonalize covariance matrix
    V, D = np.linalg.eigh(C)
    # Step 5: Sort by eigenvalues
    idx = np.argsort(-D)
    V = V[idx]
    D = D[idx]
    # Step 6: Select K components
    cum_var = np.cumsum(D / np.sum(D))
    K = np.where(cum_var >= 0.95)[0][0] + 1
    # Step 7: PCA basis
    PC = V[:, :K]
    # Step 8: Mean spectrum
    mu = np.mean(X, axis=0)
    return PC, mu, K
