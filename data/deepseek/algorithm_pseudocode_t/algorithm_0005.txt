algorithm CoreReconstructionFramework
    input: sparsePhotometryData, priorDistributions
    output: reconstructedSEDs, parameterUncertainties, classificationProbabilities

    for each object in sparsePhotometryData:
        initialize objectResults
            objectResults.reconstructedSED = empty dictionary
            objectResults.parameterUncertainties = empty dictionary
            objectResults.classificationProbabilities = empty dictionary
        
        define forwardModel(object):
            for each wavelength band in allBands:
                if band exists in sparsePhotometryData:
                    compute predictedMagnitude using forwardModel
                else:
                    predictedMagnitude = undefined or NaN
                end if
            end for
        end define
        
        define likelihoodFunction(objectSED, observedMagnitudes):
            compute probability of observedMagnitudes given objectSED
            considering bands with and without data
        end define
        
        define priorsSEDParameters:
            set prior distributions for SED parameters (e.g., albedo, phase curve amplitudes)
            based on priorDistributions
        end define
        
        perform BayesianInference:
            using MCMC or other sampler to draw samples from posterior distribution
            posterior = likelihood * priors
            collect posterior samples for SED parameters
        end perform
        
        analyzePosteriorSamples:
            compute parameterUncertainties (standard deviation of posterior samples)
            identify correlations between parameters
            assess goodness of fit (e.g., compare predicted and observed magnitudes)
        end analyze
        
        computeClassificationProbabilities:
            for each taxonomic class:
                compute probability that object belongs to this class
            end for
            identify outliers based on low classification probabilities
        end compute
        
        store results:
            objectResults.reconstructedSED = median or mean of posterior samples
            objectResults.parameterUncertainties = computed uncertainties
            objectResults.classificationProbabilities = computed probabilities
        end store
    end for
    
    output:
        return reconstructedSEDs, parameterUncertainties, classificationProbabilities
