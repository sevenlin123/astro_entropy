# 1.1 Load reference spectra
spectra_raw = load_spectra()           # shape (Ns, Nλ)

# 1.2 Pre‑process spectra (normalize, remove Sun, etc.)
spectra_norm = preprocess(spectra_raw)

# 1.3 Principal Component Analysis (PCA)
PCA_model   = PCA.fit(spectra_norm)     # returns eigenvectors V (Nλ × K), means μ
coeff_real  = PCA_model.transform(spectra_norm)   # shape (Ns, K)

# 1.4 Kernel Density Estimation (KDE) in latent space
KDE_model   = KDE.fit(coeff_real, bandwidth = h)    # continuous density over K‑dim space

# 1.5 Sample synthetic latent vectors
N_synth  = 10_000
coeff_syn = KDE_model.sample(N_synth)               # shape (N_synth, K)

# 1.6 Decode synthetic spectra
spec_syn  = PCA_model.inverse_transform(coeff_syn)   # shape (N_synth, Nλ)

# 1.7 Forward–model synthetic photometry
filters = load_filter_curves()                      # list of filter response functions
phot_syn = forward_photon(spec_syn, filters)        # shape (N_synth, Nf)

# 1.8 Inject noise into synthetic photometry
phot_syn_noisy = phot_syn + np.random.normal(0,
                    0.2 * phot_syn, size=phot_syn.shape)   # 20 % Gaussian noise

# 1.9 Assemble training set for the encoder
train_X = phot_syn_noisy                     # shape (N_synth, Nf)
train_Y = coeff_syn                          # shape (N_synth, K)

# 1.10 Train encoder (AutoML regressor)
encoder = AutoMLRegressor.train(train_X, train_Y)  # returns model capable of predicting latent coefficients

# 1.11 Train taxonomy classifier in latent space
labels   = load_taxonomy_labels()            # length Ns
clf      = AutoMLClassifier.train(coeff_real, labels)   # predicts taxonomic class

# -----------------------------------------------
# 2. INFERENCE PHASE (new observation)
# -----------------------------------------------

def reconstruct_spectrum(phot_obs):
    """
    Input:
        phot_obs : array of observed photometry [Nf]
    Output:
        spec_mean      : mean reconstructed spectrum [Nλ]
        spec_conf      : confidence band (e.g., ±1σ) [Nλ]
        tax_prob       : taxonomy probability vector
    """

    # 2.1 Predict posterior of latent coefficients
    # ---- Use encoder to get point estimate
    coeff_hat = encoder.predict(phot_obs)          # shape (K,)
    
    # ---- Sample from a Gaussian copula centered on coeff_hat
    #      (covariance estimated from encoder output distribution)
    cov = encoder.posterior_covariance(phot_obs)   # shape (K,K)
    n_samples = 500
    coeff_samples = sample_gaussian_copula(coeff_hat, cov, n_samples)
    
    # 2.2 Decode sampled latent vectors to spectra
    spec_samples = PCA_model.inverse_transform(coeff_samples)   # shape (n_samples, Nλ)
    
    # 2.3 Compute mean and confidence band
    spec_mean = np.mean(spec_samples, axis=0)                  # shape (Nλ,)
    spec_std  = np.std(spec_samples, axis=0)                   # shape (Nλ,)
    spec_conf = (spec_mean - spec_std, spec_mean + spec_std)   # tuple
    
    # 2.4 Taxonomy probabilities
    tax_prob = clf.predict_proba(coeff_hat)   # vector of class probabilities
    
    return spec_mean, spec_conf, tax_prob
