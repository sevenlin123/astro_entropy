INPUT
    S_ref[1..N]          # Reflectance spectra (high‑resolution) for N training TNOs
    λ_spec[1..L]         # Wavelength grid for spectra
    λ_filt[1..M]         # Effective wavelengths / throughput curves of the photometric filters
    N_KDE_samples        # Number of synthetic latent vectors to generate (e.g., 5×10⁴)
    h_kde                # Bandwidth for Gaussian kernel density estimation

OUTPUT
    PCA_basis[1..K]      # First K principal components (K ≤ 10)
    PCA_mean              # Mean spectrum used for centering
    KDE_model             # Kernel density estimate over K‑dimensional latent space
    Regressor            # Trained AutoML model mapping photometry → latent coefficients
    Classifier           # Trained AutoML model mapping latent mean → taxonomy probabilities

1.1 Center spectra:  S_centered[i] = S_ref[i] – mean(S_ref)
1.2 Compute covariance matrix C = cov(S_centered)
1.3 Perform eigen‑decomposition of C → eigenvalues λ_j, eigenvectors v_j
1.4 Sort eigenvectors by decreasing λ_j
1.5 Select first K eigenvectors → PCA_basis[1..K]
1.6 Project each spectrum onto PCA basis:
        coeff[i][k] = (S_centered[i] · v_k)   for k = 1..K

2.1 Fit Gaussian KDE to the set {coeff[i]} (i = 1..N):
        KDE_model = KDE(coeff, bandwidth = h_kde)
2.2 Sample synthetic latent vectors:
        for s = 1 .. N_KDE_samples
            latent_s = KDE_model.sample()
            Store latent_s

3.1 For each synthetic latent vector latent_s:
        a. Reconstruct synthetic spectrum:
               S_syn = PCA_mean + Σ_{k=1..K} latent_s[k] * v_k
        b. Convolve S_syn with each filter throughput to get ideal photometry:
               m_ideal[m] = ∫ S_syn(λ) * T_m(λ) dλ   for m = 1..M
        c. Add Gaussian noise to each band:
               σ_m = 0.20 * m_ideal[m]          # 20 % relative uncertainty
               m_noisy[m] = m_ideal[m] + Normal(0, σ_m)
        d. Store pair (m_noisy, latent_s)

4.1 Prepare training set:
        X_enc = {m_noisy}          # M‑dimensional photometry vectors
        Y_enc = {latent_s}        # K‑dimensional latent vectors
4.2 Use AutoML (e.g., AutoGluon) to train regression model:
        Regressor = AutoML.train(X_enc, Y_enc, objective = "mse")

5.1 Label each training spectrum with taxonomy T_i ∈ {H₂O, CO₂, Organics, Methanol}
5.2 Prepare classifier training data:
        X_cls = {coeff[i]}          # K‑dimensional latent means
        Y_cls = {T_i}
5.3 Train classifier with AutoML:
        Classifier = AutoML.train_classifier(X_cls, Y_cls)

INPUT
    m_obs[1..M]          # Measured photometry with uncertainties σ_obs[m]

6.1 Obtain point prediction from regressor:
        μ_lat = Regressor.predict(m_obs)
6.2 Estimate uncertainty on each latent dimension:
        # Option 1: Regressor provides variance per dimension
        # Option 2: Use ensemble predictions to build covariance Σ_lat
        Σ_lat = estimate_covariance(Regressor, m_obs)

7.1 Draw N_sample latent vectors:
        for n = 1 .. N_sample
            z_n ~ N(0, I_K)                     # Standard normal
            y_n = μ_lat + L * z_n               # Cholesky transform: Σ_lat = L*Lᵀ
            Ensure y_n lies within KDE support (e.g., reject if KDE density < ε)
            Store y_n

8.1 For each sampled latent vector y_n:
        S_n = PCA_mean + Σ_{k=1..K} y_n[k] * v_k
8.2 Compute posterior statistics:
        S_mean(λ)   = average over {S_n} for each λ
        S_lower(λ) = 2.5th percentile of {S_n}
        S_upper(λ) = 97.5th percentile of {S_n}

9.1 Pass μ_lat (or a weighted average of {y_n}) to Classifier:
        P_tax = Classifier.predict_proba(μ_lat)

OUTPUT
    S_mean(λ)          # Reconstructed mean spectrum
    [S_lower(λ), S_upper(λ)]  # Credible interval (≈95 %)
    P_tax              # Probability distribution over taxonomic classes
