# ------------------------------------------------------------------
# 1. Preparatory Stage – Build Latent Prior and Synthetic Training Set
# ------------------------------------------------------------------

# 1.1 Load reference high‑resolution TNO spectra
REF_SPECTRA = load_spectra(file_list)            # shape (N_ref, N_wave)

# 1.2 Standardise spectra (normalise by solar spectrum if needed)
STD_SPECTRA = standardise(REF_SPECTRA)

# 1.3 PCA – define low‑dimensional latent space
PCA_MODEL   = PCA(n_components = K)              # K chosen ≈ 5–8
LATENT_REAL = PCA_MODEL.fit_transform(STD_SPECTRA)      # shape (N_ref, K)

# 1.4 Kernel Density Estimation (KDE) – build continuous prior
KDE_MODEL   = KDE(kernel='gaussian', bandwidth = h)     # h ≈ 0.5
KDE_MODEL.fit(LATENT_REAL)

# 1.5 Sample synthetic latent vectors from KDE
N_SYNTH = 5000                                   # number of synthetic samples
LATENT_SYNTH = KDE_MODEL.sample(N_SYNTH)          # shape (N_SYNTH, K)

# 1.6 Reconstruct synthetic spectra from latent vectors
SYNTH_SPEC = PCA_MODEL.inverse_transform(LATENT_SYNTH)   # shape (N_SYNTH, N_wave)

# 1.7 Forward‑model synthetic photometry
FORWARD_FUNC = build_forward_model(filter_set)    # convolution with filter throughput
PHOT_SYNTH = FORWARD_FUNC(SYNTH_SPEC)             # shape (N_SYNTH, N_filters)

# 1.8 Inject realistic noise into synthetic photometry
PHOT_SYNTH_NOISY = add_gaussian_noise(PHOT_SYNTH, sigma_rel=0.20)

# 1.9 Assemble training database
TRAIN_DB = {
    'photometry' : PHOT_SYNTH_NOISY,            # inputs
    'latent'     : LATENT_SYNTH,                # targets
    'labels'     : assign_taxonomy(SYNTH_SPEC)   # optional, for classifier
}

# ------------------------------------------------------------------
# 2. Model Training Stage
# ------------------------------------------------------------------

# 2.1 Train encoder – photometry → latent coefficients
ENCODER = AutoML_Regression()
ENCODER.train(TRAIN_DB['photometry'], TRAIN_DB['latent'])

# 2.2 Train classifier – latent → taxonomy
CLASSIFIER = AutoML_Classifier()
CLASSIFIER.train(TRAIN_DB['latent'], TRAIN_DB['labels'])

# ------------------------------------------------------------------
# 3. Inference Stage – New Observation
# ------------------------------------------------------------------

def infer_spectrum(new_phot):
    """
    Input: new_phot – array of observed photometric fluxes / reflectances
            length = N_filters
    Output: 
        - posterior samples of reconstructed spectra
        - taxonomy probability vector
    """

    # 3.1 Encode photometry to latent mean & uncertainty
    latent_mean, latent_cov = ENCODER.predict_with_uncertainty(new_phot)

    # 3.2 Build Gaussian Copula from (latent_mean, latent_cov)
    COPULA = GaussianCopula(latent_mean, latent_cov)

    # 3.3 Sample M latent vectors from posterior
    M = 1000
    LATENT_SAMPLES = COPULA.sample(M)               # shape (M, K)

    # 3.4 Decode latent samples to spectra
    SPEC_SAMPLES = PCA_MODEL.inverse_transform(LATENT_SAMPLES)

    # 3.5 Classify taxonomy from latent mean
    TAX_PROB = CLASSIFIER.predict_proba(latent_mean)

    return SPEC_SAMPLES, TAX_PROB

# ------------------------------------------------------------------
# 4. Post‑processing (optional) – Summary Statistics
# ------------------------------------------------------------------

def summarize_reconstruction(SPEC_SAMPLES):
    """
    Compute median spectrum, credible intervals, spectral slopes,
    and absorption depths from posterior samples.
    """
    median_spec      = median(SPEC_SAMPLES, axis=0)
    lower_ci, upper_ci = percentile(SPEC_SAMPLES, [2.5, 97.5], axis=0)
    slope            = compute_slope(median_spec)
    absorptions      = identify_absorption_features(median_spec)
    return median_spec, lower_ci, upper_ci, slope, absorptions
