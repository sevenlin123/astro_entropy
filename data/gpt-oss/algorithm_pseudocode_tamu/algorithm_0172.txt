        Filter throughput curves {F_j(λ)}   j = 1…N_filt

STEP 1.1  // Standardise spectra
    For every S_i:
        Interpolate onto common wavelength grid λ_ref
        Normalise by continuum (e.g., divide by median reflectance)

STEP 1.2  // PCA on spectra
    [E, Λ] ← PCA(S_i)               // E : eigenvectors (K_pc ≤ 10)
    For every S_i:
        z_i ← Eᵀ · (S_i – mean_S)   // Project onto PCA basis → latent vector (length K_pc)

STEP 1.3  // Density estimation in latent space (kernel density)
    Choose KDE bandwidth h ≈ 0.5
    kde_model ← KDE(z_i , kernel = gaussian , bandwidth = h)

STEP 1.4  // Augment training set
    N_aug ← 10⁴                                // number of synthetic latent samples
    {z_syn} ← Sample(kde_model , N_aug)         // draw N_aug latent vectors

pseudocode
FOR each latent vector z in {z_syn}:
    // Reconstruct full spectrum
    S_full ← E · z + mean_S

    // Convolve with filter responses
    FOR each filter j:
        m_j ← ∫ S_full(λ) · F_j(λ) dλ / ∫ F_j(λ) dλ      // theoretical reflectance in filter j

    // Inject photometric noise (≈20 % relative uncertainty)

        σ_j ← 0.20 * m_j
        m_j_noisy ← Normal(m_j , σ_j)

END FOR
