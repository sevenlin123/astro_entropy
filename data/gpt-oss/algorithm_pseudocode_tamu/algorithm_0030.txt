# ------------------------------------------------------------------
# 1. DATA PREPARATION
# ------------------------------------------------------------------

# 1.1 Load observed high‑resolution spectra {S_i} (i = 1…N)
spectra = load_spectra()                     # array [N, L]   (L = number of wavelength bins)

# 1.2 (Optional) Normalise / continuum‑fit spectra if required
spectra_norm = normalise(spectra)            # unit‑slope or zero‑median

# 1.3 Perform Principal Component Analysis (PCA)
#    - mean subtract
S_mean = mean(spectra_norm, axis=0)
S_centered = spectra_norm – S_mean
U, Σ, VT = svd(S_centered, full_matrices=False)
#    - keep first K principal components
K = choose_K()                               # e.g. 4–5 for taxonomy, 8–10 for reconstruction
PC_basis = U[:, :K]                          # [L, K]
#    - project spectra onto latent space
latent_real = S_centered @ PC_basis          # [N, K]

# ------------------------------------------------------------------
# 2. MANIFOLD AUGMENTATION
# ------------------------------------------------------------------

# 2.1 Fit a non‑parametric Kernel Density Estimate (KDE) to latent_real
kde = fit_kde(latent_real, bandwidth=h)     # h ≈ 0.5

# 2.2 Sample M synthetic latent vectors from KDE
M = 5000                                     # number of synthetic points
latent_synth = kde.sample(M)                 # [M, K]

# 2.3 Re‑construct synthetic spectra from latent vectors
spectra_synth = (latent_synth @ PC_basis.T) + S_mean  # [M, L]

# ------------------------------------------------------------------
# 3. FORWARD MODELING (Implicit Likelihood)
# ------------------------------------------------------------------

# 3.1 Convolve synthetic spectra with instrument filter curves
filters = load_filters()                     # list of transmission curves
phot_synth_clean = convolve_with_filters(spectra_synth, filters)
#    phot_synth_clean shape: [M, B] where B = number of filters

# 3.2 Inject realistic Gaussian noise (e.g., 20 % relative)
noise_level = 0.20
phot_synth_noisy = phot_synth_clean * (1 + noise_level * randn([M, B]))

# ------------------------------------------------------------------
# 4. ENCODER TRAINING (Photometry → Latent Coefficients)
# ------------------------------------------------------------------

# 4.1 Prepare training data for AutoML regressor
X_train = phot_synth_noisy                # [M, B]
Y_train = latent_synth                    # [M, K]

# 4.2 Train an AutoML ensemble regressor (e.g., XGBoost + NN)
regressor = AutoML_Regressor()
regressor.fit(X_train, Y_train)             # learns mapping m → μ_latent

# 4.3 (Optional) Train a classifier in latent space
classifier = AutoML_Classifier()
classifier.fit(latent_real, taxonomic_labels)

# ------------------------------------------------------------------
# 5. INFERENCE (New Observation)
# ------------------------------------------------------------------

def reconstruct_spectrum(observed_photometry):
    """
    Input:
        observed_photometry : array [B]
    Output:
        spectrum_samples   : array [N_samples, L]
        spectrum_mean      : array [L]
        spectrum_confidence_interval : (low, high) arrays [L]
        taxonomy_probabilities      : dict label→prob
    """

    # 5.1 Predict mean latent coefficients (posterior mean)
    mu_latent = regressor.predict(observed_photometry)      # [K]

    # 5.2 Estimate posterior covariance of latent coefficients
    #       (e.g., from regressor ensemble variance or a separate heteroscedastic model)
    cov_latent = regressor.predict_cov(observed_photometry) # [K, K]

    # 5.3 Sample N_samples latent vectors from a Gaussian Copula
    N_samples = 1000
    #    (a) Draw standard normal samples
    Z = randn([N_samples, K])
    #    (b) Apply Cholesky of cov_latent to impose correlations
    L_chol = cholesky(cov_latent)
    latent_samples = mu_latent + Z @ L_chol.T              # [N_samples, K]

    # 5.4 Decode latent samples to spectra
    spectra_samples = (latent_samples @ PC_basis.T) + S_mean   # [N_samples, L]

    # 5.5 Compute spectrum statistics
    spectrum_mean = mean(spectra_samples, axis=0)              # [L]
    spectrum_low  = percentile(spectra_samples, 2.5, axis=0)   # 95% CI lower bound
    spectrum_high = percentile(spectra_samples, 97.5, axis=0)  # 95% CI upper bound

    # 5.6 Taxonomy probabilities
    tax_probs = classifier.predict_proba(mu_latent.reshape(1,-1))   # [1, n_classes]
    tax_dict   = dict(zip(classifier.classes_, tax_probs.ravel()))

    return {
        "samples"        : spectra_samples,
        "mean"           : spectrum_mean,
        "ci_lower"       : spectrum_low,
        "ci_upper"       : spectrum_high,
        "taxonomy_prob"  : tax_dict
    }
