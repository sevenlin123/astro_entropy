# ------------------------------------------
# 1. PRIOR BUILDING (training phase)
# ------------------------------------------

# 1.1 Load and preprocess reference spectra
S_ref = load_spectra()                  # shape: (N_spec, N_wave)
S_ref = reflectance_normalization(S_ref) # remove solar spectrum if needed
μ_S  = mean(S_ref, axis=0)               # mean spectrum
S_dc = S_ref - μ_S                       # centered spectra

# 1.2 Principal‑Component Analysis (PCA)
K   = choose_dimension()                 # e.g., 5–10 PCs
PCA = fit_pca(S_dc, n_components=K)     # returns eigenvectors V (N_wave × K)
S_lat = project(S_dc, PCA.V)             # latent coefficients, shape: (N_spec, K)

# 1.3 Kernel Density Estimate (KDE) augmentation
bandwidth = 0.5                          # optimal h
kde = fit_kde(S_lat, bandwidth=bandwidth)

N_aug = 5000                             # number of synthetic latent samples
S_lat_aug = sample_from_kde(kde, N_aug)  # shape: (N_aug, K)

# 1.4 Generate synthetic spectra from augmented latents
S_aug_dc = reconstruct(S_lat_aug, PCA.V)  # inverse PCA + mean
S_aug    = S_aug_dc + μ_S                # full spectra

# 1.5 Forward‑model synthetic photometry
filters = load_filter_curves()           # dictionary: {filter_name: throughput array}
for each synthetic spectrum s in S_aug:
    for each filter f in filters:
        m_f = convolve_with_filter(s, f)          # reflectance in that band
        m_f_noisy = add_gaussian_noise(m_f, 0.20) # 20% relative σ
    store (photometry_vector, S_lat_aug_row)      # X_train, y_train

# 1.6 Assemble training data
X_train = array of photometry vectors          # shape: (N_aug, N_filters)
y_lat   = array of latent coefficient vectors# shape: (N_aug, K)

# 1.7 Train AutoML regressor (Encoder)
regressor = auto_ml_train(X_train, y_lat)        # e.g., GradientBoostedTrees + NN

# 1.8 Train AutoML classifier (Taxonomy)
labels_ref = load_taxonomy_labels()             # shape: (N_spec,)
classifier = auto_ml_train(S_lat, labels_ref)   # supervised classification in latent space

# ------------------------------------------
# 2. INFERENCE / RECONSTRUCTION (prediction phase)
# ------------------------------------------

def reconstruct_spectrum(photometry_obs):
    """
    Input: photometry_obs – vector of observed reflectances/magnitudes in N_filters
    Output: reconstructed high‑resolution spectrum (S_rec) and taxonomy probabilities
    """

    # 2.1 Predict latent coefficients
    coeff_pred = predict(regressor, photometry_obs)            # shape: (K,)

    # 2.2 Predict taxonomy probabilities
    tax_prob   = predict(classifier, coeff_pred)              # shape: (n_classes,)

    # 2.3 Decode to full spectrum
    S_rec_dc = reconstruct(coeff_pred, PCA.V)                 # inverse PCA
    S_rec    = S_rec_dc + μ_S                                 # add mean

    return S_rec, tax_prob
