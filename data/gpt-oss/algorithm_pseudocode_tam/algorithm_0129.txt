INPUT:
    S_ref[1..N_ref]          # Reflectance spectra of N_ref ≈ 50 reference TNOs
    λ_ref[1..M]              # Wavelength grid for reference spectra
    R_j[1..F]                # Throughput curves for F photometric filters
    n_aug                     # Number of synthetic latent samples (e.g. 5000)
    h_KDE                      # KDE bandwidth (≈0.5)
    σ_phot   = 0.20            # Relative photometric noise (20%)

OUTPUT:
    For each new photometric vector m_new[1..F]:
        S_rec[1..M]           # Reconstructed spectrum
        C_prob[1..C]          # Probabilities for C taxonomic classes

--------------------------------------------------------------------
1. TRAINING PHASE
--------------------------------------------------------------------

1.1 PCA on Reference Spectra
    # Center spectra
    μ_S = mean(S_ref, axis=1)
    S_centered = S_ref - μ_S
    # Compute covariance matrix
    Cov = (S_centered * S_centered.T) / (N_ref - 1)
    # Eigen-decomposition
    eig_vals, eig_vecs = eig(Cov)
    # Sort eigenvalues descending
    idx = argsort(eig_vals)[::-1]
    eig_vals = eig_vals[idx]
    eig_vecs = eig_vecs[:, idx]
    # Choose K principal components (e.g. K=8)
    K = choose_K(eig_vals, threshold=0.99)
    PC = eig_vecs[:, :K]                # M x K matrix

1.2 Project Reference Spectra into Latent Space
    Z_ref[1..N_ref][1..K] = (S_centered.T @ PC)   # Latent coefficients

1.3 KDE Augmentation in Latent Space
    # Fit KDE to Z_ref
    kde_model = KDE(Z_ref, bandwidth=h_KDE)
    # Sample synthetic latent vectors
    Z_syn[1..n_aug][1..K] = kde_model.sample(n_aug)

1.4 Generate Synthetic Spectra
    # Reconstruct spectra from synthetic latent vectors
    S_syn[1..n_aug][1..M] = (Z_syn @ PC.T) + μ_S

1.5 Forward Model Synthetic Photometry
    For i in 1..n_aug:
        For j in 1..F:
            # Convolve synthetic spectrum with filter throughput
            m_syn[i,j] = ∫ S_syn[i](λ) * R_j(λ) dλ
        # Add Gaussian noise (relative 20%)
        m_syn[i] += Normal(0, σ_phot * m_syn[i])

1.6 Assemble Training Dataset
    X_train = m_syn       # n_aug x F photometric matrix
    Y_latent = Z_syn      # n_aug x K latent matrix
    Y_class  = class_labels(Z_syn)   # Optional: assign class using nearest ref

1.7 Train AutoML Regressor (Encoder)
    regressor = AutoML_Train(X_train, Y_latent)
        # Example: gradient‑boosted trees ensemble

1.8 Train AutoML Classifier (Latent‑Space Classifier)
    classifier = AutoML_Train(Y_latent, Y_class)
        # Outputs probability distribution over C classes

--------------------------------------------------------------------
2. INFERENCE PHASE
--------------------------------------------------------------------

For each new photometric observation m_new[1..F]:

2.1 Predict Latent Coefficients
    z_pred = regressor.predict(m_new)      # Shape: 1 x K
    # Optionally obtain predictive variance if regressor supports it

2.2 Reconstruct Spectrum (Decoder)
    S_rec = (z_pred @ PC.T) + μ_S           # Shape: 1 x M

2.3 Classify Taxonomy
    C_prob = classifier.predict_proba(z_pred)

2.4 Output
    Return S_rec, C_prob
