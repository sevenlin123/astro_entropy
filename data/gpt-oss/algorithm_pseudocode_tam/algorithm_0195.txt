# ------------------------------------------------------------------
# 1. PRE‑PROCESSING – Build the Latent Spectral Prior
# ------------------------------------------------------------------
Input:
    Spec[1..N_ref][1..M]          # Reference TNO reflectance spectra
    Filters[1..B][1..M]           # Effective throughput curves of B filters
    K                              # Desired dimensionality of latent space
    h                              # KDE bandwidth (≈0.5)

Procedure:

1.1  MeanSpectrum ← mean(Spec, axis=0)
1.2  Centered ← Spec – MeanSpectrum
1.3  PCs, Scree ← PCA(Centered, n_components=K)   # PCs: K×M matrix
1.4  Latent ← PCs · Transpose(Centered)           # N_ref × K matrix

1.5  # Construct a continuous prior via KDE
      KDE_prior ← KernelDensity(Latent, bandwidth=h)

1.6  # Generate a large synthetic latent set
      Latent_samp ← KDE_prior.sample(N_aug)       # N_aug ≫ N_ref

1.7  # Reconstruct synthetic spectra from sampled latents
      Spec_samp ← MeanSpectrum + Latent_samp · Transpose(PCs)
      # Spec_samp : N_aug × M

1.8  # Forward‑model synthetic photometry
      Phot_samp ← zeros(N_aug, B)
      for i in 1..N_aug:
          for j in 1..B:
              Flux ← integrate( Spec_samp[i] * Filters[j] ) / integrate( Filters[j] )
              Phot_samp[i,j] ← Flux
          # Add realistic noise (20 % relative reflectance)
          Phot_samp[i,:] ← Phot_samp[i,:] + Normal(0, 0.2*Phot_samp[i,:])

# ------------------------------------------------------------------
# 2. TRAINING – Learn Photometry → Latent Mapping
# ------------------------------------------------------------------
Input:
    Phot_samp (N_aug × B)        # Synthetic noisy photometry
    Latent_samp (N_aug × K)      # Corresponding latent coordinates

Procedure:

2.1  # AutoML regressor (e.g., ensemble of gradient‑boosted trees)
      Regressor ← AutoML_Train(
                     X = Phot_samp,
                     y = Latent_samp,
                     task = 'regression',
                     options = {predict_variance=True}
                  )

2.2  # AutoML classifier operating in latent space
      Classifier ← AutoML_Train(
                       X = Latent_samp,
                       y = TaxonLabel[1..N_ref],   # e.g. H₂O, CO₂, Organic, Methanol
                       task = 'classification',
                       options = {}
                    )

# ------------------------------------------------------------------
# 3. INFERENCE – Reconstruct Spectrum for New Photometry
# ------------------------------------------------------------------
Input:
    Phot_obs[1..B]              # Observed photometry (with errors)

Procedure:

3.1  # Predict latent mean and variance
      z_mean, z_var ← Regressor.predict(Phot_obs, return_variance=True)

3.2  # Draw posterior samples of latent vector
      N_post ← 1000
      Z_samples ← Normal(z_mean, sqrt(z_var), size=(N_post, K))

3.3  # Decode each latent sample into a spectrum
      Spec_samples ← MeanSpectrum + Z_samples · Transpose(PCs)   # N_post × M

3.4  # Compute posterior statistics
      Spec_mean     ← mean(Spec_samples, axis=0)
      Spec_CI_low   ← percentile(Spec_samples, 2.5, axis=0)
      Spec_CI_high  ← percentile(Spec_samples, 97.5, axis=0)

3.5  # Taxonomic probability
      Class_probs ← Classifier.predict_proba(z_mean)   # 4‑class vector

Output:
    Spec_mean          # Reconstructed reflectance spectrum
    Spec_CI_low, Spec_CI_high   # 95 % credible interval
    Class_probs        # Probabilities for H₂O, CO₂, Organic, Methanol
