Input
    SpecArray[1..N]   // High‑resolution reflectance spectra (λ_i, R_i)
    FilterCurves[1..M]// Effective throughput of M photometric filters
    Labels[1..N]      // Taxonomic class for each spectrum (optional)

Parameters
    K          // Number of principal components to retain (e.g. 8–10)
    h          // KDE bandwidth (≈0.5)
    NSamples   // Number of synthetic latent vectors (e.g. 10⁴)
    σ_noise    // Relative photometric noise level (0.20)
    EnsembleSize  // Number of models in AutoML ensemble

Procedure

1. ---- TRAINING PHASE ----------------------------------------------------
   1.1  // 1a. Compute PCA on the reference spectra
        [MeanSpec, CompMat, LatentVals] = PCA(SpecArray, K)

        // MeanSpec : (1×L) mean spectrum
        // CompMat  : (L×K) matrix of eigenvectors
        // LatentVals: (N×K) matrix of principal‑component scores

   1.2  // 1b. Estimate the latent density with KDE
        KDEmodel = FitKDE(LatentVals, bandwidth = h)

   1.3  // 1c. Generate synthetic latent vectors
        LatentSamples = Sample(KDEmodel, NSamples)   // (NSamples × K)

   1.4  // 1d. Reconstruct synthetic spectra from latent vectors
        for each ℓ in 1..NSamples
              SpecSyn[ℓ] = InversePCA(LatentSamples[ℓ], MeanSpec, CompMat)
        end for

   1.5  // 1e. Forward‑model synthetic photometry
        for each ℓ in 1..NSamples
              PhotSyn[ℓ] = ConvolveWithFilters(SpecSyn[ℓ], FilterCurves)
              // PhotSyn[ℓ] is a vector of M reflectances
        end for

   1.6  // 1f. Inject realistic photometric noise
        for each ℓ in 1..NSamples
              PhotNoisy[ℓ] = PhotSyn[ℓ] * (1 + Normal(0, σ_noise))
        end for

   1.7  // 1g. Assemble training data
        TrainingInputs  = PhotNoisy          // (NSamples × M)
        TrainingTargets  = LatentSamples     // (NSamples × K)

   1.8  // 1h. Train the encoder (latent‑to‑photometry) model
        EncoderModel = AutoML_TrainRegressor(
                           X = TrainingInputs,
                           y = TrainingTargets,
                           models = EnsembleSize )

   1.9  // 1i. Train the taxonomic classifier (latent‑to‑label)
        if Labels provided then
            ClassifierModel = AutoML_TrainClassifier(
                                 X = LatentVals,
                                 y = Labels,
                                 models = EnsembleSize )
        else
            ClassifierModel = None
        end if

2. ---- INFERENCE PHASE ----------------------------------------------------
   Input: NewPhot[1..M]   // Observed photometry (reflectance) for a target TNO

   2.1  // 2a. Predict latent coefficients with encoder
        LatPred = EncoderModel.Predict(NewPhot)    // Vector of length K
        // Optionally obtain a predictive distribution by querying
        // each member of the ensemble and aggregating.

   2.2  // 2b. Decode to full spectrum
        SpecRec = InversePCA(LatPred, MeanSpec, CompMat)

   2.3  // 2c. Classify taxonomic type
        if ClassifierModel ≠ None then
            ClassProb = ClassifierModel.PredictProba(LatPred)
        else
            ClassProb = None
        end if

   2.4  // 2d. Return results
        Output:
            Spectrum  = SpecRec
            Latent    = LatPred
            ClassProbs = ClassProb
            // Optional: uncertainty estimates from ensemble spread

End Procedure
