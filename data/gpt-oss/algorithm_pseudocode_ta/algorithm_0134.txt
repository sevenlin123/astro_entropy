# 1.1 Compute mean spectrum
mu[λ] = mean_over_N(S_train[:, λ])          # length M

# 1.2 Center training spectra
for i in 1..N:
    X[i,:] = S_train[i,:] - mu              # centered spectrum

# 1.3 Compute covariance matrix
C = (X.T @ X) / (N - 1)                    # M x M

# 1.4 Eigen‑decomposition
eig_vals, eig_vecs = eig(C)                # eig_vecs columns are PCs
sort_indices = argsort(eig_vals)[::-1]     # descending order
eig_vals_sorted = eig_vals[sort_indices]
eig_vecs_sorted = eig_vecs[:, sort_indices]

# 1.5 Select K PCs
PCs[λ, k] = eig_vecs_sorted[:, k] for k=1..K
var_exp[k] = eig_vals_sorted[k] / sum(eig_vals_sorted)

# 1.6 Project training spectra onto PCs
for i in 1..N:
    coeffs[i,k] = dot(X[i,:], PCs[:,k])     # scalar projection

# 2.1 Normalise filter responses
for j in 1..J:
    norm[j] = ∫ R_j(λ) dλ
    R_norm[j, λ] = R_j(λ) / norm[j]

# 2.2 Function to synthesize photometry from coefficients a[1..K]
function synth_photometry(a[1..K]):
    # Reconstruct spectrum
    S_rec[λ] = mu[λ] + Σ_{k=1}^{K} a[k] * PCs[λ,k]
    # Compute synthetic flux per filter
    for j in 1..J:
        p[j] = Σ_λ S_rec[λ] * R_norm[j, λ] * Δλ
    return p[1..J]

# 3.1 Prior on coefficients (zero‑mean Gaussian with variances = eigenvalues)
prior_mean[k] = 0
prior_var[k]  = eig_vals_sorted[k]               # variance per PC

# 3.2 Likelihood function
function log_likelihood(a):
    p_syn = synth_photometry(a)
    logL = 0
    for o in 1..O:
        diff = flux[o] - p_syn[obs_index[o]]
        logL += -0.5 * (diff^2 / err[o]^2 + log(2π*err[o]^2))
    return logL

# 3.3 Posterior: proportional to prior * likelihood
function log_posterior(a):
    log_prior = Σ_k -0.5 * ((a[k]-prior_mean[k])^2 / prior_var[k] + log(2π*prior_var[k]))
    return log_prior + log_likelihood(a)

# 3.4 MCMC Sampling (e.g., affine‑invariant ensemble sampler)
initialize L walkers with a random draw from prior
for iter in 1..N_iter:
    for each walker w:
        propose new a' via stretch move (as in Goodman & Weare 2010)
        accept with probability min(1, exp(log_posterior(a') - log_posterior(a_w)))
    store accepted a' into coeff_samples

# 4.1 Compute mean and covariance of coefficients
mean_coeff[k] = mean_over_L(coeff_samples[:,k])
cov_coeff[K,K] = cov(coeff_samples)

# 4.2 Reconstruct mean spectrum
spectrum[λ] = mu[λ] + Σ_k mean_coeff[k] * PCs[λ,k]

# 4.3 Propagate coefficient covariance to spectrum covariance
cov_spectrum[M,M] = 0
for λ1 in 1..M:
    for λ2 in 1..M:
        sum_val = 0
        for k1 in 1..K:
            for k2 in 1..K:
                sum_val += PCs[λ1,k1] * cov_coeff[k1,k2] * PCs[λ2,k2]
        cov_spectrum[λ1,λ2] = sum_val

# 4.4 Compute credible intervals (e.g., 16th/84th percentiles) from samples
for λ in 1..M:
    samples_spectrum[λ] = mu[λ] + Σ_k coeff_samples[:,k] * PCs[λ,k]
    lower[λ] = percentile(samples_spectrum[λ], 16)
    upper[λ] = percentile(samples_spectrum[λ], 84)
