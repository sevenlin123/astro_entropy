pseudocode
INPUT:
    S_tr  : array of shape (N_train, M)   // training spectra (wavelength samples)
    T_fil : array of shape (N_fil, M)     // filter transmission vectors
    d_obs : array of shape (N_fil)        // measured photometry
    s_err : array of shape (N_fil)        // photometric uncertainties

PARAMETERS:
    K      : number of principal components (≈4–10)
    PRIOR  : ‘gaussian’   // coefficient prior
    EPS    : numerical tolerance for matrix inversions

--------------------------------------------------------------------
# 1. Train Principal‑Component Model
--------------------------------------------------------------------
1.1 Compute mean spectrum:
        μ_s = mean(S_tr, axis=0)          # shape (M)

1.2 Center training spectra:
        X_c = S_tr – μ_s                  # shape (N_train, M)

1.3 Compute covariance matrix:
        C = (X_c^T · X_c) / (N_train-1)   # shape (M, M)

1.4 Eigen‑decomposition:
        λ, e_vec = eig(C)                 # λ: eigenvalues, e_vec: eigenvectors (columns)

1.5 Sort eigenpairs by decreasing λ.

1.6 Select first K eigenvectors:
        E_K = e_vec[:, 0:K]               # shape (M, K)

1.7 Project centered spectra onto PC space:
        coeff_train = X_c · E_K           # shape (N_train, K)

1.8 Determine prior covariance for coefficients:
        Σ_prior = diag(λ[0:K])            # shape (K, K)

--------------------------------------------------------------------
# 2. Prepare Photometric Design Matrix
--------------------------------------------------------------------
2.1 For each filter j compute its response to each PC:
        for j in 1..N_fil:
            for k in 1..K:
                B[j,k] = Σ_{m=1}^{M} ( T_fil[j,m] * E_K[m,k] ) / Σ_{m} T_fil[j,m]
    # B has shape (N_fil, K)

2.2 Compute baseline photometry from mean spectrum:
        b = Σ_{m=1}^{M} ( T_fil[j,m] * μ_s[m] ) / Σ_{m} T_fil[j,m]   for all j
    # b has shape (N_fil)

--------------------------------------------------------------------
# 3. Bayesian Inference in PC Space
--------------------------------------------------------------------
3.1 Weight matrix from measurement uncertainties:
        W = diag( 1 / (s_err ** 2) )     # shape (N_fil, N_fil)

3.2 Posterior covariance (Gaussian conjugate update):
        A = B^T · W · B + Σ_prior^{-1}  # shape (K, K)
        Σ_post = inverse(A, eps=EPS)    # use stable inversion routine

3.3 Posterior mean of coefficients:
        μ_post = Σ_post · ( B^T · W · (d_obs - b) )   # shape (K)

--------------------------------------------------------------------
# 4. Spectrum Reconstruction and Uncertainty Propagation
--------------------------------------------------------------------
4.1 Reconstructed spectrum (full resolution):
        S_rec = μ_s + E_K · μ_post          # shape (M)

4.2 Reconstructed spectrum covariance:
        Cov_S = E_K · Σ_post · E_K^T        # shape (M, M)

4.3 Credible intervals (±1.96σ):
        std_S = sqrt( diag(Cov_S) )         # shape (M)
        CI_lower = S_rec - 1.96 * std_S
        CI_upper = S_rec + 1.96 * std_S

OUTPUT:
    S_rec          // reconstructed reflectance spectrum
    std_S          // 1‑σ uncertainties per wavelength
    CI_lower, CI_upper  // 95% credible interval bounds
