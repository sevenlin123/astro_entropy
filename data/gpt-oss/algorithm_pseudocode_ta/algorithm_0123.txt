# ------------------------------------------------------------
# 1. TRAINING PHASE – Build the low‑dimensional spectral manifold
# ------------------------------------------------------------
INPUT:
    S_raw          # Matrix of N_spectra × N_wavelengths
    K_pca          # Desired number of principal components (e.g., 8–10)

PROCEDURE PCA_TRAIN(S_raw):
    # 1.1 Mean‑center the spectra
    μ_s   = mean(S_raw, axis=0)            # Shape: (N_wavelengths,)
    S_centered = S_raw - μ_s

    # 1.2 Compute the empirical covariance matrix
    C  = (S_centered.T @ S_centered) / (N_spectra - 1)   # Shape: (N_wavelengths,N_wavelengths)

    # 1.3 Eigen‑decompose and keep the leading K_pca components
    eig_vals, eig_vecs = eig(C)           # eig_vecs: columns are eigenvectors
    idx_sorted = argsort(eig_vals)[::-1]  # Descending order
    PCs = eig_vecs[:, idx_sorted[:K_pca]] # Shape: (N_wavelengths, K_pca)

    # 1.4 Project the centered spectra onto the PCs
    coeff_matrix = S_centered @ PCs       # Shape: (N_spectra, K_pca)

    RETURN μ_s, PCs, coeff_matrix
END PROCEDURE


# ------------------------------------------------------------
# 2. FORWARD MODEL – Relate PC coefficients to photometric fluxes
# ------------------------------------------------------------
INPUT:
    μ_s           # Mean spectrum (length N_wavelengths)
    PCs           # Principal components (N_wavelengths × K_pca)
    FILTERS       # List of J filters; each filter j has a transmission vector T_j
                  # T_j has length N_wavelengths (same grid as μ_s)

PROCEDURE BUILD_FORWARD_MATRIX(μ_s, PCs, FILTERS):
    J   = len(FILTERS)
    K   = PCs.shape[1]
    A   = zeros((J, K))         # Design matrix
    b   = zeros(J)              # Expected flux from mean spectrum

    FOR j IN range(J):
        T_j = FILTERS[j].transmission   # Length N_wavelengths
        # Convolve mean spectrum with filter
        b[j] = sum( μ_s * T_j )        # Simple weighted sum (approximate integration)

        # Convolve each PC with filter
        FOR k IN range(K):
            A[j,k] = sum( PCs[:,k] * T_j )
    END FOR

    RETURN A, b
END PROCEDURE


# ------------------------------------------------------------
# 3. BAYESIAN INFERENCE – Obtain posterior distribution on PC coefficients
# ------------------------------------------------------------
INPUT:
    y_obs          # Observed photometric fluxes (length J)
    σ_obs          # Measurement uncertainties (length J)
    A, b           # Forward matrix and mean spectrum contribution
    prior_cov      # Prior covariance on coefficients (K × K) – e.g., diag of sample variances

PROCEDURE BAYESIAN_INFERENCE(y_obs, σ_obs, A, b, prior_cov):
    J   = len(y_obs)
    K   = A.shape[1]

    # 3.1 Construct weighted design matrix and vector
    W   = diag( 1.0 / (σ_obs ** 2) )             # Weight matrix (J×J)
    y_w = y_obs - b                              # Residuals after subtracting mean contribution

    # 3.2 Compute posterior covariance (closed‑form for linear Gaussian)
    Σ_post = inv( A.T @ W @ A + inv(prior_cov) )  # Shape: (K,K)

    # 3.3 Compute posterior mean
    μ_post = Σ_post @ (A.T @ W @ y_w)             # Shape: (K,)

    RETURN μ_post, Σ_post
END PROCEDURE


# ------------------------------------------------------------
# 4. RECONSTRUCT FULL SPECTRUM – Generate posterior predictive spectra
# ------------------------------------------------------------
INPUT:
    μ_s          # Mean spectrum (length N_wavelengths)
    PCs          # Principal components (N_wavelengths × K_pca)
    μ_post, Σ_post  # Posterior mean and covariance of coefficients

PROCEDURE RECONSTRUCT_SPECTRUM(μ_s, PCs, μ_post, Σ_post, N_samples=1000):
    # 4.1 Draw samples of coefficients from posterior
    coeff_samples = mvn_rvs(mean=μ_post, cov=Σ_post, size=N_samples)   # Shape: (N_samples, K)

    # 4.2 Compute corresponding spectra
    spectra_samples = μ_s[None,:] + coeff_samples @ PCs.T              # Shape: (N_samples, N_wavelengths)

    # 4.3 Estimate predictive mean and uncertainty per wavelength
    spectrum_mean   = mean(spectra_samples, axis=0)                     # Shape: (N_wavelengths,)
    spectrum_std    = std(spectra_samples, axis=0)                      # Shape: (N_wavelengths,)

    RETURN spectrum_mean, spectrum_std
END PROCEDURE


# ------------------------------------------------------------
# 5. MAIN PIPELINE – Apply to one TNO with sparse photometry
# ------------------------------------------------------------
INPUT:
    S_raw, K_pca
    FILTERS
    y_obs, σ_obs
    prior_cov_factor  # scalar to scale prior covariance if desired

# 5.1 Train PCA on known spectra
μ_s, PCs, coeff_matrix = PCA_TRAIN(S_raw)

# 5.2 Build forward matrix for the chosen filters
A, b = BUILD_FORWARD_MATRIX(μ_s, PCs, FILTERS)

# 5.3 Set prior covariance (could be diagonal of coeff_matrix variance)
prior_cov = prior_cov_factor * diag( var(coeff_matrix, axis=0) )

# 5.4 Infer posterior distribution of PC coefficients
μ_post, Σ_post = BAYESIAN_INFERENCE(y_obs, σ_obs, A, b, prior_cov)

# 5.5 Reconstruct the full spectrum with uncertainty estimates
spectrum_mean, spectrum_std = RECONSTRUCT_SPECTRUM(μ_s, PCs, μ_post, Σ_post)

OUTPUT:
    spectrum_mean   # Point estimate of the TNO spectrum
    spectrum_std    # 1‑σ uncertainty band at each wavelength
