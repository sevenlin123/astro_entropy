# 1.1 Mean spectrum
μ ← mean(S_train, axis=0)                     # shape: (N_wave,)

# 1.2 Centered spectra
S_centered ← S_train – μ                       # shape: (N_spec, N_wave)

# 1.3 Covariance matrix (or use SVD directly)
C ← (S_centered.T @ S_centered) / (N_spec-1)    # shape: (N_wave, N_wave)

# 1.4 Eigen decomposition
eig_vals, eig_vecs ← eig(C)                    # eig_vecs columns = PCs
# Sort by descending eigenvalue
order ← argsort(eig_vals)[::-1]
eig_vals ← eig_vals[order]
eig_vecs ← eig_vecs[:, order]

# 1.5 Keep first k PCs
U_k ← eig_vecs[:, :k]                          # shape: (N_wave, k)
λ_k ← eig_vals[:k]                             # variances along PCs

# 2.1 Base response (mean spectrum)
r0 ← zeros(N_filter)
for j in range(N_filter):
    r0[j] ← ∫ μ(λ) * F[j](λ) dλ                # numerical integration on wavelength grid

# 2.2 PC responses
R ← zeros((N_filter, k))
for i in range(k):
    for j in range(N_filter):
        R[j,i] ← ∫ U_k[:,i](λ) * F[j](λ) dλ

# 3.1 Center observed photometry
d ← y – r0                                     # shape: (N_filter,)

# 3.2 Weight matrix (inverse variances)
W ← diag(1.0 / σ**2)                           # shape: (N_filter, N_filter)

# 3.3 Posterior covariance (analytic for Gaussian priors & likelihood)
Λ_inv ← diag(1.0 / λ_k)                        # prior precision
Sigma_post ← inv( R.T @ W @ R + Λ_inv )         # shape: (k, k)

# 3.4 Posterior mean
mu_post ← Sigma_post @ ( R.T @ W @ d )          # shape: (k,)

# 4.1 Mean reconstructed spectrum
spec_mean ← μ + U_k @ mu_post                  # shape: (N_wave,)

# 4.2 Optional: Full spectrum covariance
#   Cov(spec) = U_k @ Sigma_post @ U_k.T        # shape: (N_wave, N_wave)

# 5.1 Draw N_samples coefficient vectors from posterior
coeff_samples ← multivariate_normal(mean=mu_post,
                                   cov=Sigma_post,
                                   size=N_samples)

# 5.2 Convert to spectra
samples ← []
for coeff in coeff_samples:
    s ← μ + U_k @ coeff
    append(samples, s)
