INPUT
    S_train   : (N_spec × N_wave) matrix of training near‑IR spectra
    F_resp    : (N_filt × N_wave) matrix of filter response functions
    f_obs     : (N_filt × 1) vector of observed photometry for a target TNO
    σ_obs     : (N_filt × 1) vector of photometric uncertainties

PARAMETERS
    K         : number of principal components to retain (default 8–10)
    NSAMP     : number of posterior samples to draw (e.g. 1000)

OUTPUT
    μ_spec    : (N_wave × 1) posterior mean spectrum
    CI_low, CI_high : lower/upper 95 % credible‑interval spectra
    samp_spec : (N_wave × NSAMP) matrix of posterior predictive spectra


--------------------------------------------------------------------
1. TRAINING PHASE
--------------------------------------------------------------------
1.1 Compute the mean spectrum
        μ_s = mean(S_train, axis=0)            # shape (N_wave)

1.2 Center spectra
        S_cent = S_train – μ_s                 # (N_spec × N_wave)

1.3 Perform PCA (e.g., SVD)
        U, Σ, Vt = svd(S_cent, full_matrices=False)
        PC_basis = Vt.T[:, 0:K]                # (N_wave × K)
        λ_vals   = Σ[0:K]                      # eigenvalues for first K components

1.4 Project training spectra onto PCs
        Z = S_cent @ PC_basis                  # (N_spec × K)
        # Prior on latent vector: zero‑mean Gaussian
        S_prior = diag(λ_vals)                 # (K × K)

--------------------------------------------------------------------
2. PRECOMPUTE FILTER‑PROJECTION MATRIX
--------------------------------------------------------------------
For each filter j = 1 … N_filt
    For each component k = 1 … K
        M[j,k] = ∑_{λ} F_resp[j,λ] * PC_basis[λ,k]  # discrete integration over wavelength
End
# Mean flux per filter from mean spectrum
μ_flux = ∑_{λ} F_resp[j,λ] * μ_s[λ]  for each j   # (N_filt × 1)

--------------------------------------------------------------------
3. BAYESIAN INFERENCE FOR THE TARGET
--------------------------------------------------------------------
3.1 Observation residual
        d = f_obs – μ_flux                   # (N_filt × 1)

3.2 Measurement covariance
        C = diag(σ_obs^2)                    # (N_filt × N_filt)

3.3 Posterior covariance (analytical, linear–Gaussian)
        A = inv(S_prior^-1 + M.T @ inv(C) @ M) # (K × K)

3.4 Posterior mean
        μ_lat = A @ M.T @ inv(C) @ d          # (K × 1)

--------------------------------------------------------------------
4. POSTERIOR PREDICTION
--------------------------------------------------------------------
4.1 Sample latent vector from posterior
    For i = 1 … NSAMP
        z_i = multivariate_normal(mean=μ_lat, cov=A)
        # Reconstruct full spectrum
        s_i = μ_s + PC_basis @ z_i           # (N_wave × 1)
        samp_spec[:,i] = s_i
    End

4.2 Compute posterior summary statistics
    μ_spec = mean(samp_spec, axis=1)          # (N_wave × 1)
    CI_low , CI_high = percentile(samp_spec, [2.5,97.5], axis=1)

--------------------------------------------------------------------
RETURN
    μ_spec, CI_low, CI_high, samp_spec
