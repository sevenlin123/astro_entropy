INPUT
    • N_TNOs          # number of objects
    • N_filters       # number of photometric filters
    • F[1..N_TNOs][1..N_filters]   # measured fluxes
    • σ_F[1..N_TNOs][1..N_filters] # flux uncertainties
    • λ_eff[1..N_filters]           # effective wavelengths of filters
    • Basis[1..K][1..N_filters]     # K template spectra (normalized)

OUTPUT
    • Spectrum_recon[1..N_TNOs][1..N_filters]   # posterior mean spectra
    • Class_prob[1..N_TNOs][1..K]                # posterior class probabilities

ALGORITHM
--------------------------------------------------------------
FOR each TNO i = 1 TO N_TNOs DO

    1. // Pre‑processing
       FOR each filter j = 1 TO N_filters DO
           f_i_j      ← F[i][j]
           σ_f_i_j    ← σ_F[i][j]
       END FOR

    2. // Define parameter vector θ = {a_k}  (mixing coefficients for K bases)
       // Constraints: a_k ≥ 0, Σ a_k = 1

    3. // Construct model flux vector m(θ) for all filters
       FOR each filter j = 1 TO N_filters DO
           m_j(θ) ← Σ_{k=1}^{K} a_k * Basis[k][j]
       END FOR

    4. // Likelihood: Gaussian over observed fluxes
       LogL(θ | data) =
           -0.5 * Σ_{j=1}^{N_filters}
                 [ (f_i_j - m_j(θ))^2 / σ_f_i_j^2
                 + ln(2π σ_f_i_j^2) ]

    5. // Prior over mixing coefficients
       // Dirichlet prior with concentration α_k
       LogPrior(θ) = Σ_{k=1}^{K} (α_k - 1) * ln(a_k)

    6. // Posterior density
       LogPosterior(θ) = LogL(θ | data) + LogPrior(θ)

    7. // Sampling the posterior (e.g., Hamiltonian Monte Carlo)
       Initialize θ_0 (e.g., uniform over simplex)
       Run HMC for L iterations, storing samples {θ^(ℓ)}

    8. // Estimate posterior mean spectrum
       Spectrum_recon[i][j] = Σ_{ℓ=1}^{L} m_j(θ^(ℓ)) / L
       // For each filter j, compute mean of model fluxes over samples

    9. // Compute posterior class probability
       // For each class k, average its coefficient across samples
       FOR each class k = 1 TO K DO
           Class_prob[i][k] = (1/L) * Σ_{ℓ=1}^{L} a_k^(ℓ)
       END FOR

END FOR
--------------------------------------------------------------
RETURN Spectrum_recon, Class_prob
