Input:
    Nf          – number of photometric filters
    F[1:Nf]     – observed fluxes (or magnitudes) for one TNO
    σ[1:Nf]     – photometric uncertainties
    T[1:Nf][λ]  – filter transmission curves
    B[1:M][λ]   – spectral basis functions (e.g., PCA components)
    Solar[λ]    – solar spectrum
    μ0[1:M]     – prior mean of basis coefficients
    Σ0[1:M,1:M] – prior covariance matrix
    nSamples    – number of MCMC posterior samples to draw
    PropCov[1:M,1:M] – proposal covariance for Metropolis–Hastings

Output:
    w_samples[1:nSamples][1:M]      – posterior samples of coefficients
    S_post[λ]                       – posterior median spectrum
    CredInt[1:M]                    – 16th & 84th percentile intervals of coefficients
    ClassProb[classes]              – probability of belonging to each taxonomy class

--------------------------------------------------------------------
Step 0 – Pre‑compute design matrix A
--------------------------------------------------------------------
for i = 1 to Nf:
    for j = 1 to M:
        numerator   = ∫ B[j][λ] * T[i][λ] dλ
        denominator = ∫ Solar[λ] * T[i][λ] dλ
        A[i][j] = numerator / denominator
    end for
end for

--------------------------------------------------------------------
Step 1 – Define log‑posterior function
--------------------------------------------------------------------
function logPosterior(w[1:M]):
    # Predict fluxes from current coefficient vector
    f_pred = A * w                      # vector of length Nf

    # Log‑likelihood (Gaussian)
    logLike = 0
    for i = 1 to Nf:
        residual = F[i] - f_pred[i]
        logLike += -0.5 * ( residual² / σ[i]² + log(2π * σ[i]²) )
    end for

    # Log‑prior (multivariate normal)
    diff   = w - μ0
    logPrior = -0.5 * ( diffᵀ * inverse(Σ0) * diff )
                - 0.5 * log(det(Σ0))
                - 0.5 * M * log(2π)

    return logLike + logPrior
end function

--------------------------------------------------------------------
Step 2 – Initialise MCMC
--------------------------------------------------------------------
w_current = μ0                            # start at prior mean
logPost_current = logPosterior(w_current)
samples = empty list of size nSamples

--------------------------------------------------------------------
Step 3 – Metropolis–Hastings sampling
--------------------------------------------------------------------
for s = 1 to nSamples:
    # Propose new coefficient vector
    w_prop = w_current + Normal(0, sqrt(PropCov))   # element‑wise normal draws

    # Compute log‑posterior of proposal
    logPost_prop = logPosterior(w_prop)

    # Acceptance probability
    alpha = min(1, exp(logPost_prop - logPost_current))

    # Draw uniform random number
    u = Uniform(0,1)

    if u < alpha:
        w_current = w_prop
        logPost_current = logPost_prop
    end if

    # Store current sample
    append w_current to samples
end for

--------------------------------------------------------------------
Step 4 – Post‑processing of samples
--------------------------------------------------------------------
# Convert list to array for convenience
w_samples = array(samples)                 # shape (nSamples, M)

# Median coefficient vector (posterior median)
w_median = median(w_samples, axis=0)

# Credible intervals (16th and 84th percentiles)
CredInt = [percentile(w_samples, 16, axis=0),
           percentile(w_samples, 84, axis=0)]

# Reconstruct median spectrum
S_post[λ] = 0
for j = 1 to M:
    S_post[λ] += w_median[j] * B[j][λ]
end for

--------------------------------------------------------------------
Step 5 – Taxonomy classification (optional)
--------------------------------------------------------------------
# For each posterior sample, evaluate class likelihood
# (Assume a pre‑trained classifier that returns class probability for any spectrum)
class_counts = zeros(number_of_classes)
for s = 1 to nSamples:
    S_samp[λ] = 0
    for j = 1 to M:
        S_samp[λ] += w_samples[s][j] * B[j][λ]
    end for
    probs = Classifier(S_samp)            # vector of class probabilities
    class_counts += probs
end for
ClassProb = class_counts / nSamples

--------------------------------------------------------------------
Return:
    w_samples, w_median, CredInt, S_post, ClassProb
--------------------------------------------------------------------
