def probabilistic_spectral_reconstruction(objects, templates):
    for obj in objects:
        flux_measurements = obj.flux
        bands = obj.bands
        
        for template in templates:
            compute_expected_flux = calculate_flux_from_spectrum(template, bands)
            compute_likelihood = compute_photometric_likelihood(compute_expected_flux, flux_measurements)
            posterior_probability = bayesian_inference(compute_likelihood, template.priors)
            obj.posteriors.append((template.spectral_class, posterior_probability))
        
        classify_spectral_type = select_max_posterior_spectral_class(obj.posteriors)
        obj.spectral_type = classify_spectral_type
        
        compute_outlier_score = assess_posterior_divergence(obj.posteriors)
        obj.outlier_score = compute_outlier_score
