Algorithm CoreReconstructionFramework():
    PretrainedPCA = TrainPCA(TrainingSpectra)
    
    For each Object in Data:
        SparsePhotometry = ReadPhotometricData(Object)
        InferredComponents = BayesianInference(PretrainedPCA, SparsePhotometry)
        Uncertainties = PropagateUncertainties(InferredComponents)
        ReconstructedSpectrum = ReconstructSpectrum(InferredComponents, PretrainedPCA)
        CredibleIntervals = CalculateCredibleIntervals(ReconstructedSpectrum, Uncertainties)
        
        Output += {ReconstructedSpectrum, CredibleIntervals}
    End For
End Algorithm
